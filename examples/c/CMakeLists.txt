# Build and run the C and C++ examples against the wickra-xray C ABI.
#
#   cargo build --release -p wickra-xray-c
#   cmake -S examples/c -B examples/c/build
#   cmake --build examples/c/build --config Release
#   ctest --test-dir examples/c/build -C Release --output-on-failure
#
# WICKRA_XRAY_LIB_DIR defaults to the workspace target/release directory and can
# be overridden for out-of-tree builds.

cmake_minimum_required(VERSION 3.15)
project(wickra_xray_examples C CXX)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(WICKRA_XRAY_LIB_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../target/release"
    CACHE PATH "Directory containing the built wickra-xray C ABI library")

set(WICKRA_XRAY_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../../bindings/c/include")

find_library(WICKRA_XRAY_LIB
    NAMES wickra_xray libwickra_xray
    HINTS "${WICKRA_XRAY_LIB_DIR}"
    REQUIRED)

add_executable(frame frame.c)
target_include_directories(frame PRIVATE "${WICKRA_XRAY_INCLUDE}")
target_link_libraries(frame PRIVATE "${WICKRA_XRAY_LIB}")

add_executable(framepp frame.cpp)
target_include_directories(framepp PRIVATE "${WICKRA_XRAY_INCLUDE}")
target_link_libraries(framepp PRIVATE "${WICKRA_XRAY_LIB}")

# Golden parity from C. There is no portable directory API in C, so the spec
# list is globbed here and written into a generated header -- which keeps the
# property the other bindings get from a runtime glob: a spec added to the
# corpus is covered without editing the test. CONFIGURE_DEPENDS makes the build
# system re-glob when the directory changes, so a new spec does not need a
# manual re-configure to be noticed.
set(GOLDEN_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../golden")
file(GLOB GOLDEN_SPEC_FILES CONFIGURE_DEPENDS "${GOLDEN_DIR}/specs/*.json")
set(GOLDEN_SPEC_ENTRIES "")
foreach(spec_file ${GOLDEN_SPEC_FILES})
    get_filename_component(spec_name "${spec_file}" NAME)
    string(APPEND GOLDEN_SPEC_ENTRIES "    \"${spec_name}\",
")
endforeach()
list(LENGTH GOLDEN_SPEC_FILES GOLDEN_SPEC_COUNT)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/golden_specs.h"
"/* Generated by examples/c/CMakeLists.txt -- do not edit. */
"
"#ifndef WICKRA_XRAY_GOLDEN_SPECS_H
"
"#define WICKRA_XRAY_GOLDEN_SPECS_H
"
"#include <stddef.h>
"
"#define GOLDEN_DIR \"${GOLDEN_DIR}\"
"
"static const char *GOLDEN_SPECS[] = {
${GOLDEN_SPEC_ENTRIES}};
"
"static const size_t GOLDEN_SPEC_COUNT = ${GOLDEN_SPEC_COUNT};
"
"#endif
")

# The tests, as opposed to the examples above: `frame` and `framepp` print a
# frame and exit zero, which passes whatever the numbers say. These assert.
foreach(abi_test golden_test scrub_test)
    add_executable(${abi_test} ${abi_test}.c)
    target_include_directories(${abi_test} PRIVATE
        "${WICKRA_XRAY_INCLUDE}" "${CMAKE_CURRENT_BINARY_DIR}")
    target_link_libraries(${abi_test} PRIVATE "${WICKRA_XRAY_LIB}")
endforeach()

# Same properties through the header-only C++ wrapper, which is a second reach
# over the same ABI and needs its own proof.
add_executable(scrub_test_pp scrub_test.cpp)
target_include_directories(scrub_test_pp PRIVATE
    "${WICKRA_XRAY_INCLUDE}" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(scrub_test_pp PRIVATE "${WICKRA_XRAY_LIB}")

# On Windows the runtime DLL must sit next to the executable (there is no rpath);
# without it the loader pops a modal "DLL not found" dialog that hangs ctest. Copy
# it next to each example after build. (Unix resolves the .so/.dylib via the
# linker's embedded path, so this is Windows-only.)
if(WIN32)
    foreach(example frame framepp golden_test scrub_test scrub_test_pp)
        add_custom_command(TARGET ${example} POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different
                "${WICKRA_XRAY_LIB_DIR}/wickra_xray.dll"
                "$<TARGET_FILE_DIR:${example}>"
            VERBATIM)
    endforeach()
endif()

enable_testing()
add_test(NAME frame COMMAND frame)
add_test(NAME framepp COMMAND framepp)
add_test(NAME golden_test COMMAND golden_test)
add_test(NAME scrub_test COMMAND scrub_test)
add_test(NAME scrub_test_pp COMMAND scrub_test_pp)
# A missing runtime dependency must fail fast, never block the runner for hours.
set_tests_properties(frame framepp golden_test scrub_test scrub_test_pp
    PROPERTIES TIMEOUT 60)
